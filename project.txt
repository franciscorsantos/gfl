gfl_sistema/
│
├── app.py                 # Arquivo principal (Rotas do Flask e inicialização)
├── models.py              # Classes do banco de dados (SQLAlchemy)
├── requirements.txt       # Lista de dependências (Flask, SQLAlchemy, etc.)
│
├── instance/              # Pasta gerada pelo Flask (onde o SQLite ficará no dev)
│   └── gfl.db             # Nosso banco de dados local
│
├── static/                # Arquivos estáticos (Front-end Vanilla)
│   ├── css/
│   │   ├── style.css      # Estilos globais (cores, cards, botões)
│   │   └── print.css      # Regras específicas para impressão (@media print)
│   ├── js/
│   │   └── main.js        # Lógica de interface (modais, interações sem refresh)
│   └── img/               # Logos, ícones e assets visuais
│
└── templates/             # Arquivos HTML com Jinja2
    ├── base.html          # O layout mestre (Menu lateral, cabeçalho)
    ├── painel.html        # Dashboard principal (Saldos e alertas)
    ├── extrato.html       # Tabela de fluxo de caixa com filtros
    ├── lancamentos.html   # Tela de operações financeiras
    ├── cadastros.html     # Abas de gestão (Portadores, Plano de Contas, etc.)
    │
    └── components/        # Pequenos pedaços de HTML reutilizáveis
        ├── modal_lancamento.html
        └── modal_transferencia.html
