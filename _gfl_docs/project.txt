gfl_sistema/

├── backups/               # Diretório para armazenar backups do banco de dados

│
├── static/                # Arquivos estáticos (Front-end Vanilla)
│   ├── css/
│   │   ├── style.css      # Estilos globais (cores, cards, botões)
│   ├── js/
│   │   └── main.js        # Lógica de interface (modais, interações sem refresh)
│
└── templates/             # Arquivos HTML com Jinja2
    ├── base.html          # O layout mestre (Menu lateral, cabeçalho)
    ├── login.html         # Tela de autenticação de usuário
    ├── painel.html        # Dashboard principal (Saldos e alertas)
    ├── lancamentos.html   # Tela de operações financeiras
    ├── cadastros.html     # Abas de gestão (Portadores, Plano de Contas, etc.)
    ├── contas_a_pagar.html # Formulário para lançamento de novas contas a pagar
    ├── gerenciar_contas_a_pagar.html # Tela de gestão e baixa de parcelas
    ├── cartoes.html       # Gestão de cartões de crédito e faturas
    ├── extrato.html       # Extrato dinâmico com filtros avançados
    ├── relatorios.html    # Relatórios gerenciais (DRE, Análise de Custos)
    ├── usuarios.html      # Tela para gestão de usuários do sistema
    ├── logs.html          # Visualizador de logs de auditoria
    ├── backups.html       # Interface para gerenciamento de backups
    │
    └── components/        # Pequenos pedaços de HTML reutilizáveis
        ├── modal_lancamento.html
        └── modal_transferencia.html
        ├── modal_portador.html
        ├── modal_plano_conta.html

        ├── modal_forma_pagamento.html
